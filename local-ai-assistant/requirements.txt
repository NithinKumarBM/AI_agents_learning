
import requests

OLLAMA_URL = "http://localhost:11434/api/chat"

MODEL = "qwen3:1.7b"


def get_response(messages):

    payload = {
        "model": MODEL,
        "messages": messages,
        "stream": False
    }

    response = requests.post(
        OLLAMA_URL,
        json=payload,
        timeout=180
    )

    response.raise_for_status()

    result = response.json()

    return result["message"]["content"]
